les variables d'environnement dans PowerShell
$env:DB_USERNAME = "observa_user"
$env:DB_PASSWORD = Read-Host "Mot de passe PostgreSQL postgresql"

lancer l'application a partir du conteneur docker dans PowerShell
docker run -d `
  --name observa-notes `
  -p 8080:8080 `
  -e DB_USERNAME `
  -e DB_PASSWORD `
  -e SPRING_DATASOURCE_URL=jdbc:postgresql://host.docker.internal:5432/observa_notes `
  observa-notes:1.0

Démarrer l'application
.\mvnw.cmd spring-boot:run

Packaging
.\mvnw.cmd clean package -Dskiptests

Tester
.\mvnw.cmd test -Dtest=StudentServiceTest       

Lancer compose
docker compose up -d --build
