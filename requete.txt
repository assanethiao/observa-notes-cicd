Test 1 — Créer un étudiant (POST)
$student = @{
    firstName = "Assane"
    lastName = "Thiao"
    email = "assane.thiao@example.com"
} | ConvertTo-Json

Invoke-RestMethod `
    -Uri "http://localhost:8080/api/students" `
    -Method POST `
    -ContentType "application/json" `
    -Body $student

Test 2 — Récupérer tous les étudiants (GET)
Invoke-RestMethod `
    -Uri "http://localhost:8080/api/students" `
    -Method GET

Test 3 — Récupérer un étudiant par son ID (GET)
Invoke-RestMethod `
    -Uri "http://localhost:8080/api/students/1" `
    -Method GET

Test 4 — Modifier un étudiant (PUT)
$updatedStudent = @{
    firstName = "Assane"
    lastName = "Thiao"
    email = "assane.updated@example.com"
} | ConvertTo-Json

Invoke-RestMethod `
    -Uri "http://localhost:8080/api/students/1" `
    -Method PUT `
    -ContentType "application/json" `
    -Body $updatedStudent

Test 5 — Supprimer un étudiant (DELETE)
Invoke-RestMethod `
    -Uri "http://localhost:8080/api/students/1" `
    -Method DELETE

Test 6 — Ajouter une note
